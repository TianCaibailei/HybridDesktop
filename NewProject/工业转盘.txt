// 计算圆角正方形轨道上的精确坐标
function getRoundedSquarePoint(distance, S, R) {
  const Ls = S - 2 * R; // 直边长度
  const La = (Math.PI / 2) * R; // 圆弧长度
  const P = 4 * Ls + 4 * La; // 总周长

  let s = distance % P;
  if (s < 0) s += P; // 处理反转时的负距离
  const eps = 1e-6; 

  if (s <= Ls + eps) return { x: R + s, y: 0 };
  s -= Ls;
  if (s <= La + eps) {
    const theta = s / R;
    return { x: S - R + R * Math.sin(theta), y: R - R * Math.cos(theta) };
  }
  s -= La;
  if (s <= Ls + eps) return { x: S, y: R + s };
  s -= Ls;
  if (s <= La + eps) {
    const theta = s / R;
    return { x: S - R + R * Math.cos(theta), y: S - R + R * Math.sin(theta) };
  }
  s -= La;
  if (s <= Ls + eps) return { x: S - R - s, y: S };
  s -= Ls;
  if (s <= La + eps) {
    const theta = s / R;
    return { x: R - R * Math.sin(theta), y: S - R + R * Math.cos(theta) };
  }
  s -= La;
  if (s <= Ls + eps) return { x: 0, y: S - R - s };
  s -= Ls;
  const theta = s / R;
  return { x: R - R * Math.cos(theta), y: R - R * Math.sin(theta) };
}

// 统一的工作位指示器组件 (去除了文字，仅保留多层圆圈靶心标识，位置已移至内侧/外侧)
const WorkingPositionMarker = ({ angle, radius, pos }) => {
  if (angle !== undefined && radius !== undefined) {
    // 圆形转盘的指示器 (位于工位与圆心之间的径向上)
    return (
      <div 
        className="absolute top-1/2 left-1/2 z-0 flex items-center justify-center pointer-events-none"
        style={{ transform: `rotate(${angle}deg) translate(${radius}px) rotate(-${angle}deg)` }}
      >
        <Target size={28} className="text-cyan-500 animate-[pulse_2s_ease-in-out_infinite] drop-shadow-[0_0_8px_rgba(34,211,238,0.8)] opacity-90" />
      </div>
    );
  } else if (pos) {
    // 方形轨道的指示器 (位于连线的延长线上，即外侧)
    return (
      <div 
        className="absolute z-0 flex items-center justify-center pointer-events-none"
        style={{ left: pos.x, top: pos.y, transform: 'translate(-50%, -50%)' }}
      >
        <Target size={28} className="text-purple-500 animate-[pulse_2s_ease-in-out_infinite] drop-shadow-[0_0_8px_rgba(168,85,247,0.8)] opacity-90" />
      </div>
    );
  }
  return null;
}


// ==========================================
// 组件 1: 圆形转盘 (Circular Turntable)
// ==========================================
const CircularTurntable = () => {
  const [stationCount, setStationCount] = useState(8);
  const [step, setStep] = useState(0);
  const [playDirection, setPlayDirection] = useState(0); // 0: 停止, 1: 正转, -1: 反转
  
  // 独立状态数组：记录每个工位的状态索引
  const [stationStates, setStationStates] = useState(Array(24).fill(0));

  // --- 核心配置：工作位与起始角度 ---
  const anglePerStep = 360 / stationCount;
  // 规定0度起始位置为 18点钟方向 (CSS中 TranslateX 对应右侧3点钟，旋转90度即为底部6点钟/18点)
  const BASE_ANGLE = 90; 
  // 【外部设定】设定固定工作位逻辑索引（例如设定为第2个位置，若工位总数为8，步进45度，90+2*45=180度，即21点钟/9点方向）
  const CONFIG_WORKING_INDEX = Math.floor(stationCount * 0.25); 
  // 计算出工作位的绝对物理角度，确保每次转动都有工位精准对齐
  const workingAngle = BASE_ANGLE + CONFIG_WORKING_INDEX * anglePerStep;
  const workingRadius = 110; // 指示器位于工位(160)与圆心之间

  // 计算当前对准工作位的工位序号 (处理负数取模逻辑，对外暴露的值/方法)
  const activeStationIndex = ((CONFIG_WORKING_INDEX - step) % stationCount + stationCount) % stationCount;

  // 自动播放逻辑
  useEffect(() => {
    if (playDirection === 0) return;
    const interval = setInterval(() => setStep(s => s + playDirection), 1500);
    return () => clearInterval(interval);
  }, [playDirection]);

  const handleCountChange = (e) => {
    let val = parseInt(e.target.value) || 2;
    if (val < 2) val = 2;
    if (val > 24) val = 24;
    setStationCount(val);
    setStep(0);
    setStationStates(Array(24).fill(0)); // 重置状态
  };

  // 模拟外部调用：切换特定工位的状态
  const cycleStationState = (index) => {
    setStationStates(prev => {
      const next = [...prev];
      next[index] = (next[index] + 1) % STATION_STATES.length;
      return next;
    });
  };

  return (
    <div className="bg-slate-800 rounded-2xl p-6 shadow-2xl border border-slate-700 flex flex-col items-center">
      <div className="w-full flex justify-between items-center mb-6 pb-4 border-b border-slate-700">
        <h2 className="text-xl font-bold text-slate-100 flex items-center gap-2">
          <Activity className="text-cyan-400" />
          圆形多工位转盘
        </h2>
        <div className="flex gap-3 text-sm">
          <div className="flex items-center gap-2 bg-slate-900 px-3 py-1.5 rounded-lg border border-slate-700">
            <span className="text-slate-400">工位数</span>
            <input 
              type="number" 
              value={stationCount} 
              onChange={handleCountChange}
              className="w-12 bg-transparent text-white font-mono text-center outline-none"
            />
          </div>
        </div>
      </div>

      {/* 渲染视口 */}
      <div className="relative w-[460px] h-[460px] bg-slate-900 rounded-xl overflow-hidden shadow-inner flex items-center justify-center border border-slate-800/50">
        
        {/* 底盘背景 */}
        <div className="absolute w-[360px] h-[360px] rounded-full border-[16px] border-slate-800 bg-slate-900 shadow-[inset_0_0_40px_rgba(0,0,0,0.8)]"></div>
        <div className="absolute w-[328px] h-[328px] rounded-full border-2 border-slate-700 border-dashed opacity-50"></div>

        {/* 固定的工作位指示器 (位于工位与圆心之间) */}
        <WorkingPositionMarker angle={workingAngle} radius={workingRadius} />

        {/* 旋转的主容器 */}
        <div 
          className="absolute w-[320px] h-[320px] transition-transform duration-700 ease-in-out"
          style={{ transform: `rotate(${step * anglePerStep}deg)` }}
        >
          {/* 支撑轴臂 */}
          {Array.from({ length: stationCount }).map((_, i) => (
            <div 
              key={`spoke-${i}`} 
              className="absolute top-1/2 left-1/2 origin-left h-3 bg-gradient-to-r from-slate-600 to-slate-500 rounded-full border border-slate-500/50 shadow-md"
              style={{
                width: '160px',
                transform: `translateY(-50%) rotate(${BASE_ANGLE + i * anglePerStep}deg)`
              }} 
            />
          ))}

          {/* 中心枢纽 */}
          <div className="absolute top-1/2 left-1/2 w-28 h-28 -ml-14 -mt-14 rounded-full bg-slate-700 border-[6px] border-slate-600 shadow-[0_0_30px_rgba(0,0,0,0.6)] z-20 flex items-center justify-center">
            <div className="w-16 h-16 rounded-full bg-slate-800 border-4 border-slate-500 flex flex-col items-center justify-center shadow-inner leading-tight">
               <div className="text-[9px] text-slate-400 scale-90" title="总运动步数">POS: {step}</div>
               <div className="font-mono font-bold text-cyan-400 text-sm mt-0.5" title="当前停靠工位">ST-{activeStationIndex}</div>
            </div>
          </div>

          {/* 独立工位 */}
          {Array.from({ length: stationCount }).map((_, i) => {
            // 工位状态由内部数组维持，不随位置变化
            const statusObj = STATION_STATES[stationStates[i]];
            // 基于设定的18点钟(BASE_ANGLE)起算每个工位的物理角度
            const physicalAngle = BASE_ANGLE + i * anglePerStep;

            return (
              <div 
                key={`st-${i}`} 
                className="absolute top-1/2 left-1/2 w-16 h-16 -ml-8 -mt-8 z-30"
                style={{ transform: `rotate(${physicalAngle}deg) translate(160px) rotate(-${physicalAngle}deg)` }}
              >
                {/* 反向旋转包裹层，抵消主容器的旋转，确保文字永远水平正向 */}
                <div 
                  className="w-full h-full transition-transform duration-700 ease-in-out"
                  style={{ transform: `rotate(-${step * anglePerStep}deg)` }}
                >
                  <StationCard 
                    id={i} 
                    name={`ST-${i}`} 
                    statusObj={statusObj} 
                    onClick={cycleStationState}
                  />
                </div>
              </div>
            );
          })}
        </div>
      </div>

      {/* 控制面板 (支持正反转) */}
      <div className="w-full mt-6 flex flex-col gap-3">
        <div className="flex justify-between items-center bg-slate-900/50 p-2 rounded-lg border border-slate-700">
          <div className="flex justify-center gap-2">
            <button onClick={() => setPlayDirection(-1)} className={`ctrl-btn ${playDirection === -1 ? 'active-btn' : ''}`} title="自动反转">
              <ChevronsLeft size={18} />
            </button>
            <button onClick={() => { setStep(s => s - 1); setPlayDirection(0); }} className="ctrl-btn" title="单步反转">
              <ChevronLeft size={18} /> 逆步
            </button>
            <button onClick={() => setPlayDirection(0)} className={`ctrl-btn ${playDirection === 0 ? 'bg-slate-700 text-white' : ''}`} title="暂停">
              <Pause size={18} />
            </button>
            <button onClick={() => { setStep(s => s + 1); setPlayDirection(0); }} className="ctrl-btn" title="单步正转">
              顺步 <ChevronRight size={18} />
            </button>
            <button onClick={() => setPlayDirection(1)} className={`ctrl-btn ${playDirection === 1 ? 'active-btn' : ''}`} title="自动正转">
              <ChevronsRight size={18} />
            </button>
          </div>
          
          {/* 模拟对外获取方法的返回值 */}
          <div className="text-right pr-2">
            <div className="text-[10px] text-slate-400">API: getActiveStation()</div>
            <div className="text-sm font-bold text-cyan-400">ST-{activeStationIndex} 已对准</div>
          </div>
        </div>
      </div>
    </div>
  );
};


// ==========================================
// 组件 2: 圆角正方形转盘 (Square Track Turntable)
// ==========================================
const SquareTurntable = () => {
  const [perSideCount, setPerSideCount] = useState(3);
  const [step, setStep] = useState(0);
  const [playDirection, setPlayDirection] = useState(0);
  const [animatedStep, setAnimatedStep] = useState(0);

  // 独立状态数组
  const [stationStates, setStationStates] = useState(Array(32).fill(0)); // 最大8*4=32

  // 轨道参数
  const totalStations = perSideCount * 4;
  const TRACK_SIZE = 340;
  const TRACK_RADIUS = 80;
  const PERIMETER = 4 * (TRACK_SIZE - 2 * TRACK_RADIUS) + 2 * Math.PI * TRACK_RADIUS;
  const distancePerStation = PERIMETER / totalStations;

  // 【外部设定】设定固定工作位：距离起点的索引倍数
  // 此处设定方型轨道的工作位为初始的第0号工位位置 (即顶部直线段起点)
  const CONFIG_WORKING_INDEX = 0; 
  const workingDistance = CONFIG_WORKING_INDEX * distancePerStation;
  const workingPos = getRoundedSquarePoint(workingDistance, TRACK_SIZE, TRACK_RADIUS);

  // 计算方形工作位指示器的外侧延长线坐标
  const cx = TRACK_SIZE / 2;
  const cy = TRACK_SIZE / 2;
  const dx = workingPos.x - cx;
  const dy = workingPos.y - cy;
  const len = Math.sqrt(dx * dx + dy * dy);
  const markerOffset = 45; // 向外侧延长的距离
  const extWorkingPos = {
    x: workingPos.x + (dx / len) * markerOffset,
    y: workingPos.y + (dy / len) * markerOffset
  };

  // 计算当前对准工作位的工位序号 (对外暴露的方法)
  const activeStationIndex = ((CONFIG_WORKING_INDEX - step) % totalStations + totalStations) % totalStations;

  // 自动播放逻辑
  useEffect(() => {
    if (playDirection === 0) return;
    const interval = setInterval(() => setStep(s => s + playDirection), 1500);
    return () => clearInterval(interval);
  }, [playDirection]);

  // 平滑跟轨动画引擎
  useEffect(() => {
    let animationFrameId;
    let startTimestamp = null;
    const duration = 700; 
    const initialStep = animatedStep;
    const targetStep = step;

    if (initialStep === targetStep) return;

    const stepDiff = targetStep - initialStep;

    const animate = (timestamp) => {
      if (!startTimestamp) startTimestamp = timestamp;
      const progress = Math.min((timestamp - startTimestamp) / duration, 1);
      
      const easeProgress = progress < 0.5 
        ? 2 * progress * progress 
        : 1 - Math.pow(-2 * progress + 2, 2) / 2;

      setAnimatedStep(initialStep + stepDiff * easeProgress);

      if (progress < 1) {
        animationFrameId = requestAnimationFrame(animate);
      } else {
        setAnimatedStep(targetStep);
      }
    };

    animationFrameId = requestAnimationFrame(animate);
    return () => cancelAnimationFrame(animationFrameId);
  }, [step]); 

  const handleCountChange = (e) => {
    let val = parseInt(e.target.value) || 1;
    if (val < 1) val = 1;
    if (val > 8) val = 8;
    setPerSideCount(val);
    setStep(0);
    setAnimatedStep(0);
    setStationStates(Array(32).fill(0));
  };

  const cycleStationState = (index) => {
    setStationStates(prev => {
      const next = [...prev];
      next[index] = (next[index] + 1) % STATION_STATES.length;
      return next;
    });
  };

  return (
    <div className="bg-slate-800 rounded-2xl p-6 shadow-2xl border border-slate-700 flex flex-col items-center">
      <div className="w-full flex justify-between items-center mb-6 pb-4 border-b border-slate-700">
        <h2 className="text-xl font-bold text-slate-100 flex items-center gap-2">
          <Cpu className="text-purple-400" />
          方型柔性传输轨道
        </h2>
        <div className="flex gap-3 text-sm">
          <div className="flex items-center gap-2 bg-slate-900 px-3 py-1.5 rounded-lg border border-slate-700">
            <span className="text-slate-400">单边数</span>
            <input 
              type="number" 
              value={perSideCount} 
              onChange={handleCountChange}
              className="w-12 bg-transparent text-white font-mono text-center outline-none"
            />
            <span className="text-slate-500 border-l border-slate-700 pl-2">共 {totalStations}</span>
          </div>
        </div>
      </div>

      {/* 渲染视口 */}
      <div className="relative w-[460px] h-[460px] bg-slate-900 rounded-xl overflow-hidden shadow-inner flex items-center justify-center border border-slate-800/50">
        
        {/* 中心固定工作岛 (装饰) */}
        <div className="absolute w-[200px] h-[200px] bg-slate-800 rounded-3xl shadow-[0_0_20px_rgba(0,0,0,0.5)] border border-slate-700 flex flex-col items-center justify-center">
            <div className="text-slate-600 font-bold tracking-widest text-lg opacity-50 mb-2">控制中枢</div>
            <div className="text-purple-500/50 font-mono text-xs" title="总运动步数">POS: {step}</div>
            <div className="text-purple-400 font-bold text-lg mt-1" title="当前对准工位">SQ-{activeStationIndex}</div>
        </div>

        {/* 轨道对齐包裹层 (偏移以居中) */}
        <div className="absolute" style={{ width: TRACK_SIZE, height: TRACK_SIZE, left: (460 - TRACK_SIZE) / 2, top: (460 - TRACK_SIZE) / 2 }}>
          
          {/* 渲染物理轨道 SVG */}
          <svg className="absolute top-0 left-0 w-full h-full overflow-visible pointer-events-none">
            <rect x="0" y="0" width={TRACK_SIZE} height={TRACK_SIZE} rx={TRACK_RADIUS} fill="none" stroke="#1e293b" strokeWidth="32" />
            <rect x="0" y="0" width={TRACK_SIZE} height={TRACK_SIZE} rx={TRACK_RADIUS} fill="none" stroke="#334155" strokeWidth="12" />
            <rect x="0" y="0" width={TRACK_SIZE} height={TRACK_SIZE} rx={TRACK_RADIUS} fill="none" stroke="#475569" strokeWidth="2" strokeDasharray="6 6" />
          </svg>

          {/* 方形轨道工作位指示器 (位于连线延长线外侧) */}
          <WorkingPositionMarker pos={extWorkingPos} />

          {/* 沿着轨道运动的工位 */}
          {Array.from({ length: totalStations }).map((_, i) => {
            // 平滑运动距离 (支持正反转的负数距离)
            const currentDistance = (i + animatedStep) * distancePerStation;
            const pos = getRoundedSquarePoint(currentDistance, TRACK_SIZE, TRACK_RADIUS);
            
            // 状态从独立数组获取
            const statusObj = STATION_STATES[stationStates[i]];

            return (
              <div 
                key={`sq-st-${i}`}
                className="absolute w-16 h-16 -ml-8 -mt-8 z-30"
                style={{
                  left: pos.x,
                  top: pos.y,
                  transition: 'none' 
                }}
              >
                <StationCard 
                  id={i} 
                  name={`SQ-${i}`} 
                  statusObj={statusObj} 
                  onClick={cycleStationState} 
                />
              </div>
            );
          })}
        </div>
      </div>

      {/* 控制面板 */}
      <div className="w-full mt-6 flex flex-col gap-3">
        <div className="flex justify-between items-center bg-slate-900/50 p-2 rounded-lg border border-slate-700">
          <div className="flex justify-center gap-2">
            <button onClick={() => setPlayDirection(-1)} className={`ctrl-btn ${playDirection === -1 ? 'active-btn-purple' : ''}`} title="自动反转">
              <ChevronsLeft size={18} />
            </button>
            <button onClick={() => { setStep(s => s - 1); setPlayDirection(0); }} className="ctrl-btn" title="单步反转">
              <ChevronLeft size={18} /> 逆步
            </button>
            <button onClick={() => setPlayDirection(0)} className={`ctrl-btn ${playDirection === 0 ? 'bg-slate-700 text-white' : ''}`} title="暂停">
              <Pause size={18} />
            </button>
            <button onClick={() => { setStep(s => s + 1); setPlayDirection(0); }} className="ctrl-btn" title="单步正转">
              顺步 <ChevronRight size={18} />
            </button>
            <button onClick={() => setPlayDirection(1)} className={`ctrl-btn ${playDirection === 1 ? 'active-btn-purple' : ''}`} title="自动正转">
              <ChevronsRight size={18} />
            </button>
          </div>
          
          {/* 模拟对外获取方法的返回值 */}
          <div className="text-right pr-2">
            <div className="text-[10px] text-slate-400">API: getActiveStation()</div>
            <div className="text-sm font-bold text-purple-400">SQ-{activeStationIndex} 已对准</div>
          </div>
        </div>
      </div>
    </div>
  );
};
